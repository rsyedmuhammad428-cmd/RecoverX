cmake_minimum_required(VERSION 3.15)
project(RecoverX LANGUAGES CXX)
enable_testing()
option(RX_SANITIZE "Build with AddressSanitizer + UndefinedBehaviorSanitizer (GCC/Clang)" OFF)
if(RX_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(src/RecoverX.Native)
add_subdirectory(tests/NativeTests)
